% bjr 标识：圆角方底上三个单线小写字母 b、j、r；j 的点同时是一条向下倾斜的需求曲线与
% 一条向上倾斜的供给曲线的交点（均衡点），取「经济学」之意。本标识为讲义自行设计，
% 只取人大红这一颜色，不含也不仿照任何学校徽志。
% 坐标为 64x64 画布（单位 pt）；全部用路径绘制，不依赖字体。
% 用法：\bjrmark 或 \bjrmark[0.6]（红底白字）；\bjrmarkrev[0.6]（白底红字，放在红色底上用）。
\newcommand{\bjrmarkdraw}[3]{% #1 缩放, #2 底色, #3 字色
  \scalebox{#1}{%
  \begin{tikzpicture}[x=1pt, y=1pt,
      bjrglyph/.style={draw=#3, line width=4.6pt, line cap=round, line join=round},
      bjrcurve/.style={draw=#3, line width=1.8pt, line cap=round, opacity=0.5}]
    \fill[#2, rounded corners=13pt] (0,0) rectangle (64,64);
    % 需求曲线（左上到右下）与供给曲线（左下到右上），交于 j 的点 (38,43)
    \path[bjrcurve] (23,57) .. controls (27,50) and (32,45.5) .. (38,43)
                            .. controls (43,41) and (47,40) .. (51,39.2);
    \path[bjrcurve] (26,39) .. controls (30,39.6) and (34,41) .. (38,43)
                            .. controls (44,46.5) and (50,51.5) .. (55,57.5);
    % b：竖笔加圆腹
    \path[bjrglyph] (12,17) -- (12,48);
    \path[bjrglyph] (20.5,25.5) circle (8.5);
    % j：竖笔带左钩；点即均衡点
    \path[bjrglyph] (38,34) -- (38,14) arc[start angle=0, end angle=-90, radius=6.5];
    \fill[#3] (38,43) circle (3.6pt);
    % r：竖笔加肩
    \path[bjrglyph] (48,34) -- (48,17);
    \path[bjrglyph] (48,26) .. controls (48,31.5) and (52,34.5) .. (56.5,33);
  \end{tikzpicture}}%
}
\newcommand{\bjrmark}[1][1]{\bjrmarkdraw{#1}{rucred}{white}}
